# Host.LanguageService and Host.Driver exercise the public compiler identity.
# Compile the real Version.cpp producer against the BuildId.h generated by the
# selected NSC build; a synthetic test value would stop testing that contract.
if(NSC_COMPILER_SUITE_BUILD_ID_HEADER)
    if(NOT EXISTS "${NSC_COMPILER_SUITE_BUILD_ID_HEADER}")
        message(FATAL_ERROR
            "NSC_COMPILER_SUITE_BUILD_ID_HEADER does not exist: "
            "${NSC_COMPILER_SUITE_BUILD_ID_HEADER}")
    endif()
    set(_nsc_suite_effective_build_id_header
        "${NSC_COMPILER_SUITE_BUILD_ID_HEADER}")
    get_filename_component(_nsc_suite_build_id_basic_dir
        "${_nsc_suite_effective_build_id_header}" DIRECTORY)
    get_filename_component(_nsc_suite_build_id_namespace_dir
        "${_nsc_suite_build_id_basic_dir}" DIRECTORY)
    get_filename_component(_nsc_suite_generated_include_root
        "${_nsc_suite_build_id_namespace_dir}" DIRECTORY)
elseif(NSC_COMPILER_SUITE_NSC_EXECUTABLE)
    get_filename_component(_nsc_suite_selected_nsc_bin
        "${NSC_COMPILER_SUITE_NSC_EXECUTABLE}" DIRECTORY)
    get_filename_component(_nsc_suite_selected_nsc_build
        "${_nsc_suite_selected_nsc_bin}" DIRECTORY)
    set(_nsc_suite_inferred_build_id
        "${_nsc_suite_selected_nsc_build}/generated/nativesharp/Basic/BuildId.h")
    if(NOT EXISTS "${_nsc_suite_inferred_build_id}")
        message(FATAL_ERROR
            "The selected NSC does not expose its generated BuildId.h; set "
            "NSC_COMPILER_SUITE_BUILD_ID_HEADER explicitly. Expected: "
            "${_nsc_suite_inferred_build_id}")
    endif()
    # Keep inference local to this configure pass. Persisting it in the cache
    # would retain the previous compiler identity after selecting another NSC.
    set(_nsc_suite_effective_build_id_header
        "${_nsc_suite_inferred_build_id}")
    set(_nsc_suite_generated_include_root
        "${_nsc_suite_selected_nsc_build}/generated")
else()
    message(FATAL_ERROR
        "nsc_component_tests requires NSC_COMPILER_SUITE_NSC_EXECUTABLE or "
        "NSC_COMPILER_SUITE_BUILD_ID_HEADER to compile the real product identity")
endif()

# WindowsLibraryBuilder is a real product surface. Its executable cases use
# the product implementation, so close that implementation with the same LLVM
# C ABI selected by the configured toolchain instead of replacing it with a
# test double or silently dropping the cases.
if(NOT LLVM_DIR)
    message(FATAL_ERROR
        "nsc_component_tests requires LLVM_DIR to close the Host.Linker product dependency graph")
endif()
get_filename_component(_nsc_suite_llvm_cmake_dir "${LLVM_DIR}" ABSOLUTE)
get_filename_component(_nsc_suite_llvm_library_dir
    "${_nsc_suite_llvm_cmake_dir}/../.." ABSOLUTE)
get_filename_component(_nsc_suite_llvm_root
    "${_nsc_suite_llvm_library_dir}/.." ABSOLUTE)
find_library(NSC_COMPILER_SUITE_LLVM_C_IMPORT_LIBRARY
    NAMES LLVM-C
    HINTS "${_nsc_suite_llvm_library_dir}"
    NO_DEFAULT_PATH
    NO_CACHE
    REQUIRED)
find_file(NSC_COMPILER_SUITE_LLVM_C_RUNTIME
    NAMES LLVM-C.dll
    HINTS "${_nsc_suite_llvm_root}/bin"
    NO_DEFAULT_PATH
    NO_CACHE
    REQUIRED)
